cmake_minimum_required(VERSION 3.24)
project(leo_numerical_audit LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LEO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Original or remediated source")
get_filename_component(LEO_SOURCE_DIR "${LEO_SOURCE_DIR}" ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(LEO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LEO_BUILD_ANALYSIS OFF CACHE BOOL "" FORCE)
set(LEO_BUILD_INTEGRATION ON CACHE BOOL "" FORCE)
option(LEO_ENABLE_CSPICE "Enable direct-SPICE audit" ON)
add_subdirectory("${LEO_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/leo")
add_executable(eclipse_probe eclipse_probe.cpp)
target_link_libraries(eclipse_probe PRIVATE leo::kernel)
if(LEO_ENABLE_CSPICE)
  add_executable(numerical_audit audit.cpp)
  target_include_directories(numerical_audit PRIVATE "${LEO_SOURCE_DIR}/benchmarks/cases" "${CSPICE_INCLUDE_DIR}")
  target_link_libraries(numerical_audit PRIVATE leo::integration leo::environment "${CSPICE_LIBRARY}")
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    enable_testing()
    add_test(NAME audit_trace_json COMMAND "${Python3_EXECUTABLE}"
      "${LEO_SOURCE_DIR}/tests/audit_trace_tests.py" "$<TARGET_FILE:numerical_audit>")
  endif()
endif()
