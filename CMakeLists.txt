cmake_minimum_required(VERSION 3.24)
project(leo_dynamics VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_TESTING "Build analytical tests" OFF)
option(LEO_BUILD_INTEGRATION "Build Boost.Odeint integration adapter" ON)
option(LEO_BUILD_EXAMPLES "Build propagation example" OFF)
option(LEO_ENABLE_CSPICE "Build explicit CSPICE Sun ephemeris provider" OFF)
option(LEO_BUILD_ANALYSIS "Build trajectory postprocessing and CSV export" ON)
option(LEO_TEST_PLOTTING "Run Python Matplotlib smoke tests (requires BUILD_TESTING)" OFF)
option(LEO_ENABLE_NRLMSISE00 "Build direct NRLMSISE-00 C reference adapter" OFF)
option(LEO_ENABLE_JB2008 "Build direct external official JB2008 Fortran adapter" OFF)
option(LEO_BUILD_PYTHON_BINDINGS "Build the private pybind11 extension for a local development import" OFF)
set(LEO_PYTHON_BINDING_DIR "${CMAKE_BINARY_DIR}/python/leo_dynamics"
    CACHE PATH "Directory receiving the _core extension and the staged Python facade")
if(LEO_BUILD_PYTHON_BINDINGS)
  # The extension links the static engine targets, which therefore need PIC.
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
if(LEO_BUILD_EXAMPLES AND NOT LEO_BUILD_INTEGRATION)
  message(FATAL_ERROR "LEO_BUILD_EXAMPLES requires LEO_BUILD_INTEGRATION")
endif()
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
# Record resolved header dependencies for experiment provenance, not as user defaults.
get_target_property(_leo_eigen_includes Eigen3::Eigen INTERFACE_INCLUDE_DIRECTORIES)
set(LEO_EIGEN3_INCLUDE_DIRS "${_leo_eigen_includes}" CACHE INTERNAL "Resolved Eigen include directories")
set(LEO_EIGEN3_VERSION "${Eigen3_VERSION}" CACHE INTERNAL "Resolved Eigen version")
add_library(leo_kernel STATIC
  src/kernel/attitude.cpp src/kernel/state.cpp src/kernel/spacecraft.cpp
  src/kernel/gravity.cpp src/kernel/dynamics.cpp src/kernel/orbital.cpp
  src/kernel/radiation.cpp src/kernel/aerodynamics.cpp)
add_library(leo::kernel ALIAS leo_kernel)
target_compile_features(leo_kernel PUBLIC cxx_std_20)
target_include_directories(leo_kernel PUBLIC include)
target_link_libraries(leo_kernel PUBLIC Eigen3::Eigen)
if(LEO_ENABLE_CSPICE)
  find_package(Threads REQUIRED)
  find_path(CSPICE_INCLUDE_DIR SpiceUsr.h HINTS "${CSPICE_ROOT}/include" REQUIRED)
  find_library(CSPICE_LIBRARY NAMES cspice cspice.a libcspice.a HINTS "${CSPICE_ROOT}/lib" REQUIRED)
  add_library(leo_environment STATIC src/environment/spice.cpp)
  add_library(leo::environment ALIAS leo_environment)
  target_link_libraries(leo_environment PUBLIC leo::kernel PRIVATE "${CSPICE_LIBRARY}" Threads::Threads)
  target_include_directories(leo_environment PRIVATE "${CSPICE_INCLUDE_DIR}")
endif()
if(LEO_ENABLE_NRLMSISE00 OR LEO_ENABLE_JB2008)
  add_library(leo_atmosphere STATIC)
  add_library(leo::atmosphere ALIAS leo_atmosphere)
  target_compile_features(leo_atmosphere PUBLIC cxx_std_20)
  target_link_libraries(leo_atmosphere PUBLIC leo::kernel)
  if(LEO_ENABLE_NRLMSISE00)
    include(cmake/nrlmsise00.cmake)
  endif()
  if(LEO_ENABLE_JB2008)
    include(cmake/jb2008.cmake)
  endif()
  if(LEO_ENABLE_CSPICE)
    target_sources(leo_atmosphere PRIVATE
      src/atmosphere/coordinates.cpp src/atmosphere/weather.cpp src/atmosphere/provider.cpp)
    target_link_libraries(leo_atmosphere PUBLIC leo::environment)
  endif()
endif()
if(LEO_BUILD_INTEGRATION)
  find_package(Boost 1.74 QUIET CONFIG)
  if(NOT Boost_FOUND)
    # Upstream header-only trees need module discovery, including on CMake >=3.30.
    cmake_policy(PUSH)
    if(POLICY CMP0167)
      cmake_policy(SET CMP0167 OLD)
    endif()
    find_package(Boost 1.74 REQUIRED)
    cmake_policy(POP)
  endif()
  find_package(Threads REQUIRED) # MT-001 batch.hpp uses std::jthread/std::stop_source only.
  add_library(leo_integration STATIC src/integration/error_control.cpp)
  add_library(leo::integration ALIAS leo_integration)
  target_link_libraries(leo_integration PUBLIC leo::kernel Boost::headers Threads::Threads)
  get_target_property(_leo_boost_includes Boost::headers INTERFACE_INCLUDE_DIRECTORIES)
  set(LEO_BOOST_INCLUDE_DIRS "${_leo_boost_includes}" CACHE INTERNAL "Resolved Boost include directories")
  set(LEO_BOOST_VERSION "${Boost_VERSION}" CACHE INTERNAL "Resolved Boost version")
endif()
if(LEO_BUILD_ANALYSIS)
  add_library(leo_analysis STATIC
    src/analysis/orbital.cpp src/analysis/loads.cpp src/analysis/export.cpp)
  add_library(leo::analysis ALIAS leo_analysis)
  target_link_libraries(leo_analysis PUBLIC leo::kernel)
endif()
if(LEO_BUILD_PYTHON_BINDINGS)
  if(NOT LEO_BUILD_INTEGRATION OR NOT LEO_BUILD_ANALYSIS)
    message(FATAL_ERROR "LEO_BUILD_PYTHON_BINDINGS requires LEO_BUILD_INTEGRATION and LEO_BUILD_ANALYSIS")
  endif()
  # Explicit discovery of both development dependencies; the kernel itself keeps no
  # Python dependency and the default C++ build never reaches this block.
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG QUIET)
  if(NOT pybind11_FOUND)
    # pybind11 ships its CMake package inside its own environment; ask the selected
    # interpreter so a virtual environment is used without extra configuration.
    execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir
                    OUTPUT_VARIABLE _leo_pybind11_dir OUTPUT_STRIP_TRAILING_WHITESPACE
                    RESULT_VARIABLE _leo_pybind11_result ERROR_QUIET)
    if(NOT _leo_pybind11_result EQUAL 0)
      message(FATAL_ERROR "pybind11 not found: pass -Dpybind11_DIR=<prefix>/share/cmake/pybind11")
    endif()
    find_package(pybind11 CONFIG REQUIRED PATHS "${_leo_pybind11_dir}" NO_DEFAULT_PATH)
  endif()
  pybind11_add_module(_core MODULE
    bindings/module.cpp bindings/configuration.cpp bindings/integration.cpp bindings/analysis.cpp)
  target_include_directories(_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/bindings")
  target_link_libraries(_core PRIVATE leo::integration leo::analysis)
  if(TARGET leo_environment) # Optional CSPICE provider; never required by v0.1.
    target_link_libraries(_core PRIVATE leo::environment)
  endif()
  set_target_properties(_core PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${LEO_PYTHON_BINDING_DIR}"
    RUNTIME_OUTPUT_DIRECTORY "${LEO_PYTHON_BINDING_DIR}"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  # Stage the facade and its type information beside the extension, so that
  # PYTHONPATH=<build>/python makes "import leo_dynamics" resolve this package.
  file(MAKE_DIRECTORY "${LEO_PYTHON_BINDING_DIR}")
  foreach(_leo_python_file IN ITEMS __init__.py __init__.pyi)
    # configure_file tracks input changes: facade-only edits are staged even
    # when no C++ source needs rebuilding.
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/python/leo_dynamics/${_leo_python_file}"
                   "${LEO_PYTHON_BINDING_DIR}/${_leo_python_file}" COPYONLY)
  endforeach()
  message(STATUS "leo_dynamics: development import with PYTHONPATH=${LEO_PYTHON_BINDING_DIR}/..")
endif()
if(LEO_BUILD_EXAMPLES)
  add_executable(propagate_leo examples/propagate_leo.cpp)
  target_link_libraries(propagate_leo PRIVATE leo::integration)
  if(LEO_BUILD_ANALYSIS)
    add_executable(analyze_orbit examples/analyze_orbit.cpp)
    target_link_libraries(analyze_orbit PRIVATE leo::integration leo::analysis)
    if(LEO_ENABLE_CSPICE)

    add_executable(propagate_environment examples/propagate_environment.cpp)
    target_link_libraries(propagate_environment PRIVATE leo::integration leo::environment)
  endif()
endif()

  if(LEO_ENABLE_CSPICE)
    add_executable(leo_spice_tests tests/spice_tests.cpp)

  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME audit_metric_identities COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/sensitivity_metrics_tests.py")
    add_test(NAME process_batch_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_batch_tests.py")
  endif()

    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME plotting_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/plotting_smoke.py")
